# 1. PARAMÈTRES DU SYSTÈME
# ==========================================

SERIAL_PORT = 'COM8'
BAUD_RATE = 9600

MODEL_PATH = "trained.tflite"
LABELS = ["Baffula", "Nature", "Porc"]
DISPLAY_NAMES = {"Baffula": "Buffalo", "Porc": "Pig", "Nature": "Nature"}

SENDER_EMAIL = "omarmelah59@gmail.com"
SENDER_PASSWORD = "pkej kjis rwvi wzdb"
RECEIVER_EMAIL = "oumaradoummellah@gmail.com"

EMAIL_COOLDOWN = 15
last_email_time = 0

DETECTION_COOLDOWN = 3.0
last_detection_time = 0

# Variables globales d'état
buffalo_count = 0
pig_count = 0
current_distance = 0.0
current_status = "SYSTEM ARMED & MONITORING"
status_color = "#2ECC71"

led_red_active = False

# ==========================================

interpreter = tf.lite.Interpreter(model_path=MODEL_PATH)

height = input_details[0]['shape'][1]
width = input_details[0]['shape'][2]
input_type = input_details[0]['dtype']

arduino = None
try:
    arduino = serial.Serial(SERIAL_PORT, BAUD_RATE, timeout=0.1)
    time.sleep(1.5)
    arduino.reset_input_buffer()
except Exception:
    arduino = None

cap = cv2.VideoCapture(0, cv2.CAP_DSHOW)
if not cap.isOpened():
    cap = cv2.VideoCapture(0)

# ==========================================
# 3. ENVOI DES EMAILS D'ALERTE
# ==========================================

def send_alert_email_thread(animal_type, confidence_score, image_frame):
    global last_email_time
    current_time = time.time()
    if current_time - last_email_time < EMAIL_COOLDOWN:
        return

    try:
        timestamp_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        msg = MIMEMultipart()
        msg['Subject'] = f"🚨 [SECURITY ALERT] Farm Guard: {animal_type.upper()} Detected!"
        msg['From'] = SENDER_EMAIL
        msg['To'] = RECEIVER_EMAIL

        body = f"""==================================================
           FARM GUARD SYSTEM - SECURITY REPORT
==================================================
Target Identified : {animal_type}
Confidence Level  : {confidence_score:.2f}%
Timestamp         : {timestamp_str}
Location          : Sector 1 (Main Perimeter)
Hardware Action   : Buzzer Activated & Red LED Triggered
=================================================="""
        msg.attach(MIMEText(body, 'plain'))

        _, img_encoded = cv2.imencode('.jpg', image_frame)
        image_attachment = MIMEImage(img_encoded.tobytes(), name=f"incident_{animal_type}.jpg")
        msg.attach(image_attachment)

        server = smtplib.SMTP('smtp.gmail.com', 587)
        server.starttls()
        server.login(SENDER_EMAIL, SENDER_PASSWORD)
        server.send_message(msg)
        server.quit()
        last_email_time = current_time
    except Exception:
        pass

# 4. TRAITEMENT EN ARRIÈRE-PLAN
# ==========================================

def background_worker():
    global latest_frame, last_detection_time, buffalo_count, pig_count, current_status, status_color
    global led_green_active, led_red_active, buzzer_active, current_distance, manual_override

    while is_running:
        if cap.isOpened():
            ret, frame = cap.read()
            if ret and frame is not None:
                latest_frame = frame.copy()

        if arduino and arduino.in_waiting > 0:
            try:
                raw_data = arduino.readline().decode('utf-8', errors='ignore').strip()
            except Exception:
                raw_data = ""

            if raw_data.startswith("DIST:"):
                try:
                    current_distance = float(raw_data.split(":")[1])
                except Exception:
                    pass

            # Si le mode manuel n'est pas actif, on traite les détections automatiques
            elif not manual_override:
                if raw_data == "CHECK" and latest_frame is not None:
                    current_time = time.time()
                    if current_time - last_detection_time >= DETECTION_COOLDOWN:
                        last_detection_time = current_time

                        img_resized = cv2.resize(latest_frame, (width, height))
                        img_rgb = cv2.cvtColor(img_resized, cv2.COLOR_BGR2RGB)

                        if input_type == np.int8:
                            input_data = (img_rgb.astype(np.int16) - 128).astype(np.int8)
                        elif input_type == np.uint8:
                            input_data = np.expand_dims(img_rgb.astype(np.uint8), axis=0)
                        else:
                            input_data = np.expand_dims(img_rgb.astype(np.float32) / 255.0, axis=0)

                        if input_type == np.int8:
                            input_data = np.expand_dims(input_data, axis=0)

                        interpreter.set_tensor(input_details[0]['index'], input_data)
                        interpreter.invoke()
                        output_data = interpreter.get_tensor(output_details[0]['index'])[0]

                        if output_details[0]['dtype'] in [np.int8, np.uint8]:
                            scale, zero_point = output_details[0]['quantization']
                            if scale > 0:
                                output_data = (output_data.astype(np.float32) - zero_point) * scale

                        top_idx = np.argmax(output_data)
                        raw_label = LABELS[top_idx] if top_idx < len(LABELS) else "Nature"
                        label = DISPLAY_NAMES.get(raw_label, raw_label)
                        confidence = output_data[top_idx] * 100 if output_data[top_idx] <= 1.0 else output_data[top_idx]

                        if label in ["Buffalo", "Pig"]:
                            current_status = f"🚨 INTRUSION: {label.upper()} ({confidence:.1f}%)"
                            status_color = "#E74C3C"
                            led_green_active = False
                            led_red_active = True
                            buzzer_active = True

                            if label == "Buffalo": buffalo_count += 1
                            elif label == "Pig": pig_count += 1

                            try: arduino.write(b"ALERT\n")
                            except Exception: pass
                            threading.Thread(target=send_alert_email_thread, args=(label, confidence, latest_frame)).start()
                        else:
                            current_status = f"🌿 CLEAR ZONE (Nature {confidence:.1f}%)"
                            status_color = "#2ECC71"
                            led_green_active = True
                            led_red_active = False
                            buzzer_active = False
                            try: arduino.write(b"NORMAL\n")
                            except Exception: pass

                elif raw_data == "CLEAR_DISTANCE":
                    current_status = "SYSTEM ARMED & MONITORING"
                    status_color = "#2ECC71"
                    led_green_active = True
                    led_red_active = False
                    buzzer_active = False
                    try: arduino.write(b"NORMAL\n")
                    except Exception: pass

        time.sleep(0.01)

    # --- ACTIONS MANUELLES ---
    def manual_trigger_alarm(self):
        global current_status, status_color, led_green_active, led_red_active, buzzer_active, manual_override
        manual_override = True
        current_status = "⚠️ MANUAL OVERRIDE: ALARM ACTIVE"
        status_color = "#E74C3C"
        led_green_active = False
        led_red_active = True
        buzzer_active = True
        if arduino:
            try: arduino.write(b"ALERT\n")
            except Exception: pass

    def manual_reset_normal(self):
        global current_status, status_color, led_green_active, led_red_active, buzzer_active, manual_override
        manual_override = False
        current_status = "SYSTEM ARMED & MONITORING"
        status_color = "#2ECC71"
        led_green_active = True
        led_red_active = False
        buzzer_active = False
        if arduino:
            try: arduino.write(b"NORMAL\n")
            except Exception: pass

    def manual_mute_buzzer(self):
        global buzzer_active
        buzzer_active = False
        if arduino:
            try: arduino.write(b"MUTE\n")
            except Exception: pass

    def update_gui(self):
        global latest_frame
        
        self.status_text.configure(text=current_status)
        self.status_card.configure(fg_color=status_color)
        
        if current_distance > 0:
            self.lbl_distance.configure(text=f"📡 Ultrasonic Range: {current_distance:.1f} cm")
        else:
            self.lbl_distance.configure(text="📡 Ultrasonic Range: Clear")

        # Mise à jour de la couleur des indicateurs matériels
        self.btn_green_led.configure(fg_color="#2ECC71" if led_green_active else "#2C3E50")
        self.btn_red_led.configure(fg_color="#E74C3C" if led_red_active else "#2C3E50")
        self.btn_buzzer.configure(fg_color="#E67E22" if buzzer_active else "#2C3E50")

        self.lbl_buffalo.configure(text=f"🐃 Buffalo Intrusions : {buffalo_count}")
        self.lbl_pig.configure(text=f"🐖 Pig Intrusions : {pig_count}")

        if latest_frame is not None:
            frame_rgb = cv2.cvtColor(latest_frame, cv2.COLOR_BGR2RGB)
            frame_resized = cv2.resize(frame_rgb, (680, 510))
            img_pil = Image.fromarray(frame_resized)
            img_tk = ImageTk.PhotoImage(image=img_pil)
            self.video_label.configure(image=img_tk)
            self.video_label.image = img_tk

        self.after(30, self.update_gui)

    def on_close(self):
        global is_running
        is_running = False
        if arduino:
            try:
                arduino.write(b"NORMAL\n")
                arduino.close()
            except Exception: pass
        cap.release()
        self.destroy()
